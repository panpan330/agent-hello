class LLMQueryRewriter:
    """LLM 驱动查询改写：把口语提问规范化为检索问法。失败回退规则。"""
    def __init__(self, settings: Settings, *, client: Any | None = None) -> None:
        self._settings = settings
        self._client = client
        self._fallback = RuleBasedQueryRewriter()

    def rewrite(self, query: str) -> QueryRewriteResult:
        if not self._settings.has_llm_api_key:
            return self._fallback.rewrite(query)
        try:
            reply = self._call_llm(query)  # prompt: 改写为规范检索问法，只输出改写结果
            return QueryRewriteResult(
                original_query=query, rewritten_query=reply.strip(),
                changed=reply.strip() != query, rewrite_reasons=["llm_rewrite"],
            )
        except Exception:
            return self._fallback.rewrite(query)

class LLMMultiQueryGenerator:
    """LLM 驱动多查询扩展：从语义/政策/场景/关键词角度生成检索变体。失败回退规则。"""
    def __init__(self, settings, *, client=None): self._settings = settings; self._fallback = RuleBasedMultiQueryGenerator()
    def generate(self, query, *, max_queries=DEFAULT_MULTI_QUERY_LIMIT) -> MultiQueryExpansion:
        if not self._settings.has_llm_api_key: return self._fallback.generate(query, max_queries=max_queries)
        try:
            reply = self._call_llm(query)  # prompt: 生成 max_queries 条检索变体，JSON 数组
            queries = json.loads(reply)  # 解析
            return MultiQueryExpansion(original_query=query,
                queries=[MultiQueryCandidate(query=q, query_type="llm", reason="llm") for q in queries[:max_queries]],
                expanded=len(queries) > 1, ...)
        except Exception:
            return self._fallback.generate(query, max_queries=max_queries)
